  <script>
    // Global variable to store current customer
    let currentCustomer = "";
    let isAdmin = false;
    
    function showPage(pageId) {
      // Hide all pages
      document.querySelectorAll('.page').forEach(page => {
        page.classList.remove('active');
      });
      
      // Show selected page
      document.getElementById(pageId).classList.add('active');
    }
    
    function showCustomerOverview(customerName, logoSrc) {
      // Set current customer
      currentCustomer = customerName;
      
      // Update all customer name displays
      document.getElementById('overview-customer-name').textContent = customerName;
      document.getElementById('scorecard-customer-name').textContent = customerName + " Score card";
      document.getElementById('projects-customer-name').textContent = customerName + " Project updates";
      document.getElementById('audit-customer-name').textContent = customerName + " Audit status";
      
      // Update edit buttons visibility based on admin status
      updateUIForUserRole();
      
      // Show customer overview
      showPage('customer-overview');
      
      // Load data from storage for this customer
      loadCustomerData(customerName);
    }

    // Handle login functionality
    function handleLogin() {
      const username = document.querySelector('#login-page input[type="text"]').value;
      const password = document.querySelector('#login-page input[type="password"]').value;
      
      // Very simple authentication - in a real app, this would be server-side
      if(username === "admin" && password === "admin123") {
        isAdmin = true;
        alert("Logged in as administrator");
      } else {
        isAdmin = false;
        alert("Logged in as regular user");
      }
      
      // Return to home page after login
      showPage('home-page');
      
      // Update UI based on login status
      updateUIForUserRole();
    }

    // Update UI elements based on user role
    function updateUIForUserRole() {
      const adminElements = document.querySelectorAll('.admin-only');
      adminElements.forEach(el => {
        el.style.display = isAdmin ? 'block' : 'none';
      });
      
      // Change login button to logout if logged in
      const loginButtons = document.querySelectorAll('.action-btn');
      loginButtons.forEach(btn => {
        if(isAdmin) {
          btn.textContent = 'Logout (Admin)';
          btn.onclick = function() { logout(); };
        } else {
          btn.textContent = 'Login';
          btn.onclick = function() { showPage('login-page'); };
        }
      });
    }

    // Logout function
    function logout() {
      isAdmin = false;
      updateUIForUserRole();
      showPage('home-page');
    }

    // Initialize customer data if not exists
    function initializeCustomerData() {
      if(!localStorage.getItem('customerData')) {
        const initialData = {
          'General Motors': {
            scorecard: {
              rating: 'A',
              details: 'Standard performance metrics are good'
            },
            projects: [
              {name: 'Project X', status: 'In Progress', timeline: 'Q1 2024 - Q3 2024'},
              {name: 'Project Y', status: 'Planning', timeline: 'Q3 2024 - Q1 2025'}
            ],
            audit: {
              total: 80,
              withoutOPL: 65,
              withOPL: 15,
              recent: [
                {date: '03/10/2024', type: 'Process Audit', status: 'Completed without OPL'}
              ]
            }
          }
        };
        
        localStorage.setItem('customerData', JSON.stringify(initialData));
      }
    }

    // Save customer data (for admin use)
    function saveCustomerData(customerName, section, data) {
      if(!isAdmin) return false;
      
      const allData = JSON.parse(localStorage.getItem('customerData') || '{}');
      
      if(!allData[customerName]) {
        allData[customerName] = {};
      }
      
      allData[customerName][section] = data;
      localStorage.setItem('customerData', JSON.stringify(allData));
      return true;
    }

    // Retrieve customer data
    function getCustomerData(customerName, section) {
      const allData = JSON.parse(localStorage.getItem('customerData') || '{}');
      
      if(allData[customerName] && allData[customerName][section]) {
        return allData[customerName][section];
      }
      
      return null;
    }

              <p>Status: ${project.status}</p>

            </div>
          `;
        });
        
        // Insert after the image
        const projectImage = projectsContainer.querySelector('img');
        if(projectImage) {
          projectImage.insertAdjacentHTML('afterend', projectsHTML);
        }
      }
      
      // Load audit data
      const auditData = getCustomerData(customerName, 'audit');
      if(auditData) {
        // Update audit UI with the data
        if(auditData.total) {
          document.querySelector('.audit-status p').textContent = `Total number of audit: ${auditData.total}`;
        }
        
        if(auditData.withoutOPL) {
          document.querySelector('.without-opl .stat-number').textContent = auditData.withoutOPL;
        }
        
        if(auditData.withOPL) {
          document.querySelector('.with-opl .stat-number').textContent = auditData.withOPL;
        }
        
        // Update recent audits table
        if(auditData.recent && auditData.recent.length > 0) {
          const tbody = document.querySelector('#customer-audit table tbody');
          tbody.innerHTML = '';
          
          auditData.recent.forEach(audit => {

              </tr>
            `;
            tbody.innerHTML += row;
          });
        }
      }
    }

    // Add edit functionality for admin
    function showAdminEditor(section) {
      if(!isAdmin) return;
      
      const customerName = currentCustomer;
      const data = getCustomerData(customerName, section);
      
      // Show the appropriate editor based on section
      switch(section) {
        case 'scorecard':
          showScorecardEditor(customerName, data);
          break;
        case 'projects':
          showProjectsEditor(customerName, data);
          break;
        case 'audit':
          showAuditEditor(customerName, data);
          break;
      }
    }

    // Show scorecard editor
    function showScorecardEditor(customerName, data) {
      document.getElementById('editor-title').textContent = `Edit ${customerName} Scorecard`;
      
      // Show scorecard editor, hide others
      document.getElementById('scorecard-editor').style.display = 'block';
      document.getElementById('projects-editor').style.display = 'none';
      document.getElementById('audit-editor').style.display = 'none';
      
      // Populate with existing data
      if(data) {
        document.getElementById('scorecard-rating').value = data.rating || 'A';
        document.getElementById('scorecard-details').value = data.details || '';
      }
      
      // Setup save handler
      document.getElementById('save-edit-btn').onclick = function() {
        const updatedData = {
          rating: document.getElementById('scorecard-rating').value,
          details: document.getElementById('scorecard-details').value
        };
        
        saveCustomerData(customerName, 'scorecard', updatedData);
        closeEditor();
        loadCustomerData(customerName); // Refresh displayed data
      };
      
      // Show modal
      document.getElementById('admin-editor-modal').style.display = 'block';
    }
