    // Update UI elements based on user role
    function updateUIForUserRole() {
      const adminElements = document.querySelectorAll('.admin-only');
      adminElements.forEach(el => {
        el.style.display = isAdmin ? 'block' : 'none';
      });
      
      // Change login button to logout if logged in

      loginButtons.forEach(btn => {
        if(isAdmin) {
          btn.textContent = 'Logout (Admin)';
          btn.onclick = function() { logout(); };
        } else {

          btn.onclick = function() { showPage('login-page'); };

    // Save customer data (for admin use)
    function saveCustomerData(customerName, section, data) {
      if(!isAdmin) return false;
      
      const allData = JSON.parse(localStorage.getItem('customerData') || '{}');
      
      if(!allData[customerName]) {
        allData[customerName] = {};
      }
      
      allData[customerName][section] = data;
      localStorage.setItem('customerData', JSON.stringify(allData));
      return true;
    }

    // Retrieve customer data
    function getCustomerData(customerName, section) {
      const allData = JSON.parse(localStorage.getItem('customerData') || '{}');
      
      if(allData[customerName] && allData[customerName][section]) {
        return allData[customerName][section];
      }
      
      return null;
    }

    // Load customers list
    function loadCustomersList() {
      const listContainer = document.getElementById('customers-list');
      listContainer.innerHTML = '';
      
      const allData = JSON.parse(localStorage.getItem('customerData') || '{}');
      
      // Add IATF customers
      const iatfSection = document.querySelector('#home-page .customer-section:nth-child(1) .customer-grid');
      const iatfCustomers = Array.from(iatfSection.querySelectorAll('.customer-card')).map(card => ({
        name: card.querySelector('.customer-name').textContent,
        logo: card.querySelector('.customer-logo').src.split('/').pop()
      }));
      
      // Add Non-IATF customers
      const nonIatfSection = document.querySelector('#home-page .customer-section:nth-child(2) .customer-grid');
      const nonIatfCustomers = Array.from(nonIatfSection.querySelectorAll('.customer-card')).map(card => ({
        name: card.querySelector('.customer-name').textContent,
        logo: card.querySelector('.customer-logo').src.split('/').pop()
      }));
      
      // Add to list
      iatfCustomers.forEach(customer => {
        addCustomerToEditorList('iatf', customer.name, customer.logo);
      });
      
      nonIatfCustomers.forEach(customer => {
        addCustomerToEditorList('non-iatf', customer.name, customer.logo);
      });
    }

    function addCustomer(type, name, logo) {
      const section = type === 'iatf' ?

        document.querySelector('#home-page .customer-section:nth-child(2) .customer-grid');

    }

    // Remove customer
    function removeCustomer(name) {
      // Remove from both sections
      const cards = document.querySelectorAll('.customer-card');
      cards.forEach(card => {
        if(card.querySelector('.customer-name').textContent === name) {
          card.remove();
        }
      });
      
      // Remove from storage
      const allData = JSON.parse(localStorage.getItem('customerData') || '{}');
      delete allData[name];
      localStorage.setItem('customerData', JSON.stringify(allData));
    }

    // Generate year options function (helper)
    function generateYearOptions() {
      const currentYear = new Date().getFullYear();
      let options = '';
      
      for(let year = currentYear; year >= currentYear - 10; year--) {
        options += `<option value="${year}">${year}</option>`;
      }
      
      return options;
    }

    // Helper function to read file as array buffer
    function readFileAsArrayBuffer(file) {
      return new Promise((resolve, reject) => {
        const reader = new FileReader();
        reader.onload = () => resolve(reader.result);
        reader.onerror = reject;
        reader.readAsArrayBuffer(file);
      });
    }

    function downloadProjectReport() {
      const reportData = JSON.parse(localStorage.getItem('currentProjectReport'));
      if (reportData) {
        const blob = new Blob([new Uint8Array(reportData.data)], { type: 'application/pdf' });
        const url = URL.createObjectURL(blob);
        const a = document.createElement('a');
        a.href = url;
