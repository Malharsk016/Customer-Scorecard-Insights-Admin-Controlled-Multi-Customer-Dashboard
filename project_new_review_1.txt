  <script>
    // Global variables to store current customer info
    let currentCustomer = "";
    let currentLogo = "";
    
    function showPage(pageId) {
      // Hide all pages
      document.querySelectorAll('.page').forEach(page => {
        page.classList.remove('active');
      });
      
      // Show selected page
      document.getElementById(pageId).classList.add('active');
    }
    
    function showCustomerDashboard(customerName, logoSrc) {
      // Set current customer and logo
      currentCustomer = customerName;
      currentLogo = logoSrc;
      
      // Update all customer name displays
      document.getElementById('customer-name-dashboard').textContent = customerName;
      document.getElementById('customer-name-scorecard').textContent = customerName + " Score card";
      document.getElementById('customer-name-projects').textContent = customerName + " Project updates";
      document.getElementById('customer-name-audit').textContent = customerName + " Audit status";
      
      // Update all logo displays
      document.getElementById('dashboard-logo').src = logoSrc;
      document.getElementById('scorecard-logo').src = logoSrc;
      document.getElementById('projects-logo').src = logoSrc;
      document.getElementById('audit-logo').src = logoSrc;
      
      // Show dashboard
      showPage('customer-dashboard');
    }
    
    function showCustomerPage(pageType) {
      // Show appropriate page based on type
      showPage('customer-' + pageType);
    }
  </script>
