        }
      } catch (e) {

        }

    // Add PDF to customer data
    function addCustomerPDF(customerName, pdfFile) {
      const allData = JSON.parse(localStorage.getItem('customerData') || '{}');
      
      if(!allData[customerName]) {
        allData[customerName] = {};
      }
      
      const reader = new FileReader();
      reader.onload = function(e) {
        allData[customerName].pdf = {
          name: pdfFile.name,
          data: e.target.result.split(',')[1],
          lastUpdated: new Date().toISOString()
        };
        localStorage.setItem('customerData', JSON.stringify(allData));
        alert(`PDF uploaded for ${customerName}`);
      };
      reader.readAsDataURL(pdfFile);
    }

    // Save customer data (for admin use)
    function saveCustomerData(customerName, section, data) {
      if(!isAdmin) return false;
      
      const allData = JSON.parse(localStorage.getItem('customerData') || '{}');
      
      if(!allData[customerName]) {
        allData[customerName] = {};
      }
      
      // Handle PDF file storage
      if(section === 'projects' || section === 'audit') {
        const fileInput = document.getElementById(`${section}-pdf`);
        if(fileInput && fileInput.files.length > 0) {
          const file = fileInput.files[0];
          const reader = new FileReader();
          reader.onload = function(e) {
            data.pdfData = e.target.result;
            data.pdfFileName = file.name;
            data.lastUpdated = new Date().toISOString();
            
            allData[customerName][section] = data;
            localStorage.setItem('customerData', JSON.stringify(allData));
          };
          reader.readAsDataURL(file);
          return true;
        }
      }
      
      allData[customerName][section] = data;
      localStorage.setItem('customerData', JSON.stringify(allData));
      return true;
    }

    // Retrieve customer data
    function getCustomerData(customerName, section) {
      const allData = JSON.parse(localStorage.getItem('customerData') || '{}');
      
      if(allData[customerName] && allData[customerName][section]) {
        return allData[customerName][section];
      }
      
      return null;
    }

    // View PDF in new tab
    function viewPDF(customerName, section) {
      const data = getCustomerData(customerName, section);
      if(data && data.pdfData) {
        const win = window.open();
        win.document.write('<iframe width="100%" height="100%" src="' + data.pdfData + '"></iframe>');
      }
    }

    // Download PDF
    function downloadPDF(customerName, section) {
      const data = getCustomerData(customerName, section);
      if(data && data.pdfData && data.pdfFileName) {
        const link = document.createElement('a');
        link.href = data.pdfData;
        link.download = data.pdfFileName;
        link.click();
      }
    }

    // Remove customer
    function removeCustomer(name) {
      // Remove from both sections
      const cards = document.querySelectorAll('.customer-card');
      cards.forEach(card => {
        if(card.querySelector('.customer-name').textContent === name) {
          card.remove();
        }
      });
      
      // Remove from storage
      const allData = JSON.parse(localStorage.getItem('customerData') || '{}');
      delete allData[name];
      localStorage.setItem('customerData', JSON.stringify(allData));
    }

    // Generate year options function (helper)
    function generateYearOptions() {
      const currentYear = new Date().getFullYear();
      let options = '';
      
      for(let year = currentYear; year >= currentYear - 10; year--) {
        options += `<option value="${year}">${year}</option>`;
      }
      
      return options;
    }

    // Download project report
    function downloadProjectReport() {
      const reportData = JSON.parse(localStorage.getItem('currentProjectReport'));
      if (reportData) {
        const blob = new Blob([new Uint8Array(reportData.data)], { type: 'application/pdf' });
        const url = URL.createObjectURL(blob);
        const a = document.createElement('a');
        a.href = url;
        a.download = reportData.name;
        a.click();
        URL.revokeObjectURL(url);
      }
    }
